- [ ] **Step 3: Write `seed-factors.ts`**

Create `packages/database/src/seed-factors.ts`:

```typescript
import { readFileSync } from 'fs'
import { parse } from 'csv-parse/sync'
import { db, emissionFactors } from './index'
import { sql } from 'drizzle-orm'

// ---------------------------------------------------------------------------
// Exported helpers (also used by tests)
// ---------------------------------------------------------------------------

export function parseRequiredBy(raw: string | null | undefined): string[] {
  if (!raw) return []
  const s = raw.trim()
  if (s.startsWith('{') && s.endsWith('}')) {
    return s.slice(1, -1).split(',').map(x => x.trim().replace(/^"|"$/g, '')).filter(Boolean)
  }
  return []
}

export function mapFactorType(factorType: string): { unitClass: string | null; inputUnitCanonical: string | null } {
  switch (factorType) {
    case 'combustion':       return { unitClass: 'energy',        inputUnitCanonical: 'MMBtu' }
    case 'electricity-grid': return { unitClass: 'energy',        inputUnitCanonical: 'MWh'   }
    case 'refrigerant-gwp':  return { unitClass: 'dimensionless', inputUnitCanonical: null    }
    case 'scope3-spend':     return { unitClass: 'currency',      inputUnitCanonical: 'USD'   }
    case 'scope3-distance':  return { unitClass: 'distance',      inputUnitCanonical: 'km'    }
    default:                 return { unitClass: null,             inputUnitCanonical: null    }
  }
}

// ---------------------------------------------------------------------------
// CSV file paths
// ---------------------------------------------------------------------------

const CSV_FILES = [
  'C:/Users/nimab/Neuvetra/factors/processed/combustion_epa-cfr98-2025.csv',
  'C:/Users/nimab/Neuvetra/factors/processed/electricity_grid_egrid-2023.csv',
  'C:/Users/nimab/Neuvetra/factors/processed/refrigerant_gwp_ipcc-ar6.csv',
  'C:/Users/nimab/Neuvetra/factors/processed/scope3_spend_epa-scghg-v130.csv',
  'C:/Users/nimab/Neuvetra/factors/processed/scope3_transport_defra-2024.csv',
]

// ---------------------------------------------------------------------------
// Main seed function
// ---------------------------------------------------------------------------

async function seedFactors() {
  let totalInserted = 0

  for (const csvPath of CSV_FILES) {
    const content = readFileSync(csvPath, 'utf-8')
    const rows: Record<string, string>[] = parse(content, {
      columns: true,
      skip_empty_lines: true,
    })

    const values = rows.map(r => {
      const { unitClass, inputUnitCanonical } = mapFactorType(r.factor_type)
      return {
        factorId:            r.factor_id,
        name:                r.name,
        factorType:          r.factor_type,
        scope:               r.scope ? Number(r.scope) : null,
        scope3Category:      r.scope3_category ? Number(r.scope3_category) : null,
        substance:           r.substance || null,
        value:               r.value,
        unit:                r.unit,
        co2Factor:           r.co2_factor || null,
        ch4Factor:           r.ch4_factor || null,
        n2oFactor:           r.n2o_factor || null,
        geography:           r.geography || null,
        jurisdiction:        r.jurisdiction || null,
        gwpBasis:            r.gwp_basis || null,
        tier:                r.tier ? Number(r.tier) : null,
        uncertaintyPct:      r.uncertainty_pct || null,
        sourceDocument:      r.source_document,
        sourceUrl:           r.source_url || null,
        dataYear:            r.data_year ? Number(r.data_year) : null,
        publishedDate:       r.published_date || null,
        effectiveStart:      r.effective_start,
        effectiveEnd:        r.effective_end || null,
        supersededBy:        r.superseded_by || null,
        wikiMethodPage:      r.wiki_method_page || null,
        lastVerified:        r.last_verified || null,
        requiredBy:          parseRequiredBy(r.required_by),
        unitClass,
        inputUnitCanonical,
      }
    })

    // Upsert in batches of 200 to avoid query size limits
    const BATCH = 200
    for (let i = 0; i < values.length; i += BATCH) {
      const batch = values.slice(i, i + BATCH)
      await db
        .insert(emissionFactors)
        .values(batch)
        .onConflictDoUpdate({
          target: emissionFactors.factorId,
          set: {
            name:               sql`excluded.name`,
            factorType:         sql`excluded.factor_type`,
            scope:              sql`excluded.scope`,
            scope3Category:     sql`excluded.scope3_category`,
            substance:          sql`excluded.substance`,
            value:              sql`excluded.value`,
            unit:               sql`excluded.unit`,
            co2Factor:          sql`excluded.co2_factor`,
            ch4Factor:          sql`excluded.ch4_factor`,
            n2oFactor:          sql`excluded.n2o_factor`,
            geography:          sql`excluded.geography`,
            jurisdiction:       sql`excluded.jurisdiction`,
            gwpBasis:           sql`excluded.gwp_basis`,
            tier:               sql`excluded.tier`,
            uncertaintyPct:     sql`excluded.uncertainty_pct`,
            sourceDocument:     sql`excluded.source_document`,
            sourceUrl:          sql`excluded.source_url`,
            dataYear:           sql`excluded.data_year`,
            publishedDate:      sql`excluded.published_date`,
            effectiveStart:     sql`excluded.effective_start`,
            effectiveEnd:       sql`excluded.effective_end`,
            supersededBy:       sql`excluded.superseded_by`,
            wikiMethodPage:     sql`excluded.wiki_method_page`,
            lastVerified:       sql`excluded.last_verified`,
            requiredBy:         sql`excluded.required_by`,
            unitClass:          sql`excluded.unit_class`,
            inputUnitCanonical: sql`excluded.input_unit_canonical`,
          },
        })
    }

    console.log(`[seed] ${csvPath.split('/').pop()}: ${rows.length} rows`)
    totalInserted += rows.length
  }

  console.log(`[seed] Done. Total: ${totalInserted} rows`)
}

// Only run if called directly (not imported by tests)
if (import.meta.path === Bun.main) {
  seedFactors().catch(e => { console.error(e); process.exit(1) })
}
```

```typescript
import { CalculationError } from './errors'
import type { CalculationResult } from './types'

export function validate(result: CalculationResult, allowNegative = false): void {
  if (!isFinite(result.value)) {

  }
}
```

```typescript
import { db, emissionFactors } from '@terrascope/database'
import { eq, and, isNull, lte, gt, or, sql } from 'drizzle-orm'
import type { FactorQuery, FactorReference, FactorResolver } from './types'
import { FactorNotFoundError } from './errors'

async function resolve(query: FactorQuery): Promise<FactorReference> {
  const { factorType, substance, geography, regulatoryContext, reportingYear, scope3Category } = query

  const conditions: ReturnType<typeof eq>[] = [eq(emissionFactors.factorType, factorType)]

  if (substance)        conditions.push(eq(emissionFactors.substance, substance))
  if (geography)        conditions.push(eq(emissionFactors.geography, geography))
  if (scope3Category != null) conditions.push(eq(emissionFactors.scope3Category, scope3Category))

  if (regulatoryContext) {
    conditions.push(
      sql`${emissionFactors.requiredBy} @> ARRAY[${regulatoryContext}]::text[]` as any
    )
  }

  if (reportingYear != null) {
    const yearEnd   = `${reportingYear}-12-31`
    const yearStart = `${reportingYear}-01-01`
    conditions.push(lte(emissionFactors.effectiveStart, yearEnd) as any)
    conditions.push(
      or(isNull(emissionFactors.effectiveEnd), gt(emissionFactors.effectiveEnd, yearStart)) as any
    )
  }

  const rows = await db.select().from(emissionFactors).where(and(...conditions))

  if (rows.length === 0) {
    throw new FactorNotFoundError(
      `No factor: type=${factorType} substance=${substance ?? '*'} ` +
      `geo=${geography ?? '*'} context=${regulatoryContext ?? '*'} year=${reportingYear ?? '*'}`
    )
  }

  const sorted = [...rows].sort((a, b) =>
    (b.effectiveStart ?? '').localeCompare(a.effectiveStart ?? '')
  )

  if (sorted.length > 1 && sorted[0].effectiveStart === sorted[1].effectiveStart) {
    const ids = sorted.slice(0, 3).map(r => r.factorId).join(', ')
    throw new FactorNotFoundError(`Ambiguous: multiple factors share effective_start: ${ids}`)
  }

  const r = sorted[0]
  return {
    factorId:      r.factorId,
    name:          r.name,
    value:         Number(r.value),
    unit:          r.unit,
    sourceDocument: r.sourceDocument,
    effectiveStart: r.effectiveStart,
    effectiveEnd:   r.effectiveEnd,
    gwpBasis:       r.gwpBasis,
    uncertaintyPct: r.uncertaintyPct !== null ? Number(r.uncertaintyPct) : null,
  }
}

export const dbFactorResolver: FactorResolver = { resolve }
```

```typescript
    unit: OUTPUT_UNIT,
    methodologyId: METHODOLOGY_ID,
```

```typescript
    uncertaintyPct: gwp.uncertaintyPct ?? undefined,
```

```typescript
import { convert } from '../unit-registry'
import { validate } from '../validator'
import type { CalculationContext, CalculationResult, FactorReference } from '../types'

const METHODOLOGY_ID = 'scope-2-market-based'
const METHODOLOGY_VERSION = 1
const OUTPUT_UNIT = 'kg CO2e'

interface Inputs {
  contracted_consumption: number
  contracted_unit: string
  contracted_factor_kg_per_mwh: number
  uncovered_consumption: number
  uncovered_unit: string
  grid_geography: string
  residual_mix_factor_kg_per_mwh?: number
}

export async function calculate(inputs: Inputs, context: CalculationContext): Promise<CalculationResult> {
  const contractedMwh = convert(inputs.contracted_consumption, inputs.contracted_unit, 'energy', 'MWh')
  const uncoveredMwh  = convert(inputs.uncovered_consumption,  inputs.uncovered_unit,  'energy', 'MWh')

  const factors: FactorReference[] = []
  const warnings: string[] = []
  let residualFactorValue: number

  if (inputs.residual_mix_factor_kg_per_mwh != null) {
    residualFactorValue = inputs.residual_mix_factor_kg_per_mwh
    factors.push({
      factorId: `user-supplied-residual-${inputs.grid_geography}`,
      name: `User-supplied residual mix for ${inputs.grid_geography}`,
      value: residualFactorValue,
      unit: 'kg CO2e / MWh',
      sourceDocument: 'user input',
    })
    warnings.push(`Residual-mix factor supplied by user (${residualFactorValue} kg CO2e/MWh) — not from DB.`)
  } else {
    const gridFactor = await context.factorResolver.resolve({
      factorType: 'electricity-grid',
      geography: inputs.grid_geography,
      regulatoryContext: context.regulatoryContext,
      reportingYear: context.reportingYear,
    })
    residualFactorValue = gridFactor.value
    factors.push(gridFactor)
    warnings.push(
      `True residual-mix not available for ${inputs.grid_geography}; ` +
      `using location-based grid average (${gridFactor.factorId}) as documented fallback.`
    )
  }

  factors.push({
    factorId: 'user-supplied-contracted-instrument',
    name: 'Contractual instrument emission rate (user-supplied)',
    value: inputs.contracted_factor_kg_per_mwh,
    unit: 'kg CO2e / MWh',
    sourceDocument: 'user input — contractual instrument disclosure',
  })

  const contractedEmissions = contractedMwh * inputs.contracted_factor_kg_per_mwh
  const uncoveredEmissions  = uncoveredMwh  * residualFactorValue
  const value = contractedEmissions + uncoveredEmissions

  const result: CalculationResult = {
    value,
    unit: OUTPUT_UNIT,
    methodologyId: METHODOLOGY_ID,
    methodologyVersion: METHODOLOGY_VERSION,
    factors,
    inputs: {
      contracted_mwh: contractedMwh,
      contracted_factor_kg_per_mwh: inputs.contracted_factor_kg_per_mwh,
      contracted_emissions_kg: contractedEmissions,
      uncovered_mwh: uncoveredMwh,
      residual_factor_kg_per_mwh: residualFactorValue,
      uncovered_emissions_kg: uncoveredEmissions,
      grid_geography: inputs.grid_geography,
    },
    regulatoryContext: context.regulatoryContext,
    boundaryBasis: context.boundaryMethod,
    warnings,
    computedAt: new Date().toISOString(),
  }

  validate(result)
  return result
}
```

```typescript
import type { CalculationResult } from './types'

const SCOPE2_PAIR = new Set(['scope2_location', 'scope2_market'])

interface Entry {
  bucket: string
  result: CalculationResult
  boundaryMultiplier: number
}

export class Inventory {
  private entries: Entry[] = []
  private _warnings: string[] = []

  constructor(
    public readonly boundaryMethod: string,
    public readonly regulatoryContext: string,
    public readonly reportingYear: number,
    public readonly gwpBasis = 'AR6',
    public readonly boundaryMultiplier = 1.0,
  ) {}

  addScope1(result: CalculationResult, multiplierOverride?: number): void {
    this._add('scope1', result, multiplierOverride)
  }
  addScope2Location(result: CalculationResult, multiplierOverride?: number): void {
    this._add('scope2_location', result, multiplierOverride)
  }
  addScope2Market(result: CalculationResult, multiplierOverride?: number): void {
    this._add('scope2_market', result, multiplierOverride)
  }
  addScope3(result: CalculationResult, multiplierOverride?: number): void {
    this._add('scope3', result, multiplierOverride)
  }

  private _add(bucket: string, result: CalculationResult, override?: number): void {
    if (result.regulatoryContext !== this.regulatoryContext) {
      this._warnings.push(
        `${result.methodologyId}: regulatory_context '${result.regulatoryContext}' ` +
        `doesn't match inventory '${this.regulatoryContext}'`
      )
    }

    const ids = result.factors.map(f => f.factorId)
    for (const e of this.entries) {
      if (e.bucket === bucket) continue
      if (SCOPE2_PAIR.has(e.bucket) && SCOPE2_PAIR.has(bucket)) continue
      const shared = ids.filter(id => e.result.factors.some(f => f.factorId === id))
      if (shared.length > 0) {
        this._warnings.push(
          `Double-count risk: [${shared.join(', ')}] in ${e.bucket} and ${bucket}`
        )
      }
    }

    const mult = override ?? this.boundaryMultiplier
    this.entries.push({ bucket, result, boundaryMultiplier: mult })
  }

  private bucketTotal(bucket: string): number {
    return this.entries
      .filter(e => e.bucket === bucket)
      .reduce((s, e) => s + e.result.value * e.boundaryMultiplier, 0)
  }

  scope1Total(): number          { return this.bucketTotal('scope1') }
  scope2LocationTotal(): number  { return this.bucketTotal('scope2_location') }
  scope2MarketTotal(): number    { return this.bucketTotal('scope2_market') }
  scope3Total(): number          { return this.bucketTotal('scope3') }

  grandTotal(scope2: 'market' | 'location' = 'market'): number {
    const s2 = scope2 === 'market' ? this.scope2MarketTotal() : this.scope2LocationTotal()
    return this.scope1Total() + s2 + this.scope3Total()
  }

  auditTrail(): object[] {
    return this.entries.map(e => ({
      bucket: e.bucket,
      boundaryMultiplier: e.boundaryMultiplier,
      result: e.result,
    }))
  }

  warnings(): string[] { return [...this._warnings] }

  summary(): object {
    return {
      boundaryMethod: this.boundaryMethod,
      regulatoryContext: this.regulatoryContext,
      reportingYear: this.reportingYear,
      gwpBasis: this.gwpBasis,
      scope1KgCO2e: this.scope1Total(),
      scope2LocationKgCO2e: this.scope2LocationTotal(),
      scope2MarketKgCO2e: this.scope2MarketTotal(),
      scope3KgCO2e: this.scope3Total(),
      grandTotalMarketKgCO2e: this.grandTotal('market'),
      grandTotalLocationKgCO2e: this.grandTotal('location'),
      warnings: this._warnings,
    }
  }
}
```

```typescript
import { Elysia, t } from "elysia"
import { db } from "@terrascope/database"
import { emissionFactors } from "@terrascope/database"
import { eq, and, isNull, lte, gt, or, sql } from "drizzle-orm"

export const factorsRoutes = new Elysia({ prefix: "/factors" })
  .get(
    "/",
    async ({ query }) => {
      const { type, geography, scope, scope3_category, regulatory_context, effective_at } = query

      const conditions: any[] = []

      if (type)             conditions.push(eq(emissionFactors.factorType, type))
      if (geography)        conditions.push(eq(emissionFactors.geography, geography))
      if (scope)            conditions.push(eq(emissionFactors.scope, Number(scope)))
      if (scope3_category)  conditions.push(eq(emissionFactors.scope3Category, Number(scope3_category)))

      if (regulatory_context) {
        conditions.push(
          sql`${emissionFactors.requiredBy} @> ARRAY[${regulatory_context}]::text[]`
        )
      }

      if (effective_at) {
        const year = Number(effective_at)
        const yearEnd   = `${year}-12-31`
        const yearStart = `${year}-01-01`
        conditions.push(lte(emissionFactors.effectiveStart, yearEnd))
        conditions.push(
          or(isNull(emissionFactors.effectiveEnd), gt(emissionFactors.effectiveEnd, yearStart))
        )
      } else {
        // Default: only currently active factors
        conditions.push(isNull(emissionFactors.effectiveEnd))
      }

      const factors = await db
        .select()
        .from(emissionFactors)
        .where(conditions.length > 0 ? and(...conditions) : undefined)
        .limit(100)

      return { factors }
    },
    {
      query: t.Object({
        type:               t.Optional(t.String()),
        geography:          t.Optional(t.String()),
        scope:              t.Optional(t.String()),
        scope3_category:    t.Optional(t.String()),
        regulatory_context: t.Optional(t.String()),
        effective_at:       t.Optional(t.String()),
      }),
    }
  )
  .get(
    "/:factorId",
    async ({ params }) => {
      const [factor] = await db
        .select()
        .from(emissionFactors)
        .where(eq(emissionFactors.factorId, params.factorId))
        .limit(1)
      return factor ?? { error: "Factor not found" }
    },
    {
      params: t.Object({ factorId: t.String() }),
    }
  )
```

- [ ] **Step 2: Create `apps/api/src/routes/reports.ts`**

```typescript
import { Elysia, t } from "elysia"
import { db, ghgReports } from "@terrascope/database"
import { eq } from "drizzle-orm"
import {
  dbFactorResolver,
  calculateScope1Combustion,
  calculateScope1Fugitive,
  calculateScope2Location,
  calculateScope2Market,
  Inventory,
  type CalculationContext,
} from "@terrascope/calculator"

export const reportsRoutes = new Elysia({ prefix: "/reports" })
  .post("/", async ({ body }) => {
    const {
      company_id,
      reporting_year,
      regulatory_context,
      boundary_method,
      emission_data,
    } = body

    const ctx: CalculationContext = {
      regulatoryContext: regulatory_context,
      reportingYear: reporting_year,
      gwpBasis: "AR6",
      boundaryMethod: boundary_method,
      factorResolver: dbFactorResolver,
    }

    const inv = new Inventory(boundary_method, regulatory_context, reporting_year)

    // Process each emission_data entry
    for (const entry of emission_data as any[]) {
      switch (entry.methodology) {
        case "scope-1-stationary-combustion":
          inv.addScope1(await calculateScope1Combustion(entry.inputs, ctx))
          break
        case "scope-1-fugitive-refrigerants":
          inv.addScope1(await calculateScope1Fugitive(entry.inputs, ctx))
          break
        case "scope-2-location-based":
          inv.addScope2Location(await calculateScope2Location(entry.inputs, ctx))
          break
        case "scope-2-market-based":
          inv.addScope2Market(await calculateScope2Market(entry.inputs, ctx))
          break
      }
    }

    const summary = inv.summary() as any
    const [report] = await db.insert(ghgReports).values({
      companyId:            company_id,
      reportingYear:        reporting_year,
      regulatoryContext:    regulatory_context,
      boundaryMethod:       boundary_method,
      scope1Total:          String(summary.scope1KgCO2e / 1000),
      scope2LocationBased:  String(summary.scope2LocationKgCO2e / 1000),
      scope2MarketBased:    String(summary.scope2MarketKgCO2e / 1000),
      auditTrail:           inv.auditTrail(),
    }).returning()

    return {
      report,
      summary,
      audit_trail: inv.auditTrail(),
    }
  }, {
    body: t.Object({
      company_id:         t.String(),
      reporting_year:     t.Number(),
      regulatory_context: t.String(),
      boundary_method:    t.String(),
      emission_data:      t.Array(t.Any()),
    }),
  })
  .get("/:id", async ({ params }) => {
    const [report] = await db.select().from(ghgReports).where(eq(ghgReports.id, params.id))
    return report ?? { error: "Report not found" }
  }, { params: t.Object({ id: t.String() }) })
```
